You will be redirected to the main page within 3 seconds. If not, please [click here]({{ site.baseurl | prepend: site.url }}).


<script>
  (function () {
    // Determine language from the URL
    const isChinese = /^\/zh-cn(?:\/|$)/i.test(window.location.pathname);

    // Destination homepage
    const homeURL = isChinese ? "/zh-cn/" : "/";

    // Preserve the original 404 layout
    const description = document.querySelector(".post-description");
    const paragraph = document.querySelector(".post article p");
    const link = paragraph?.querySelector("a");

    // Update language-specific text
    if (description) {
      description.textContent = isChinese
        ? "前面的区域，以后再来探索吧！"
        : "How about we explore the area ahead of us later?";
    }

    if (paragraph && link) {
      link.href = homeURL;
      link.textContent = isChinese ? "点这里" : "click here";

      paragraph.replaceChildren(
        document.createTextNode(
          isChinese
            ? "您将在3秒后传送回主页。失效的话，请"
            : "You will be redirected to the main page within 3 seconds. If not, please "
        ),
        link,
        document.createTextNode(isChinese ? "。" : ".")
      );
    }

    // Automatic redirect after 3 seconds
    setTimeout(function () {
      window.location.replace(homeURL);
    }, 3000);
  })();
</script>
